1 let & const vs var

var is function-scoped and hoisted — it leaks out of blocks. let and const are block-scoped and safe.

Example 1 — var leaks out of an if block (the bug)
if (true) {
  var leaky = "I escaped the block!";
}
console.log(leaky); // var doesn't care about the {} boundary
I escaped the block!
Example 2 — let stays in its block (the fix)
if (true) {
  let safe = "I'm block-scoped";
}
console.log(safe); // ReferenceError: safe is not defined
ReferenceError: safe is not defined
Example 3 — var leaks out of a for loop
for (var i = 0; i < 3; i++) { /* loop body */ }
console.log(i); // var i survives the loop — it's now 3!
3
Example 4 — let stays in the for loop
for (let j = 0; j < 3; j++) { /* loop body */ }
console.log(j); // ReferenceError: j is not defined
ReferenceError: j is not defined
Example 5 — const prevents reassignment (TypeError)
const MAX = 100;
MAX = 200; // TypeError: Assignment to constant variable
TypeError: Assignment to constant variable.
Example 6 — const ALLOWS mutation: change object property (works!)
const user = { name: "Alice", age: 25 };
user.age = 26;       // ✅ mutating is fine — the binding didn't change
user.city = "London"; // ✅ adding a new key is also fine
console.log(user);
{ name: 'Alice', age: 26, city: 'London' }
Example 7 — const ALLOWS mutation: push to array (works!)
const fruits = ["apple", "banana"];
fruits.push("mango"); // ✅ mutating contents is allowed
fruits[0] = "kiwi";   // ✅ changing index is allowed
console.log(fruits);
['kiwi', 'banana', 'mango']
Example 8 — var allows redeclaration: a silent, dangerous bug
var score = 10;
// ... 200 lines later, a junior dev writes...
var score = 99; // No error! Original value is quietly overwritten
console.log(score);
99
Example 9 — let blocks redeclaration: safe, explicit error
let score = 10;
let score = 99; // SyntaxError: Identifier 'score' has already been declared
SyntaxError: Identifier 'score' has already been declared
Example 10 — The classic setTimeout + var bug: all print 3
// With var — all callbacks share the SAME i variable
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// By the time callbacks run, loop is done — i is already 3
3 3 3
Example 11 — The fix with let: each iteration gets its own i
// With let — each loop creates a NEW block-scoped i
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// Each callback captured its own snapshot of i
0 1 2
Example 12 — The golden rule: when to use which
// ✅ Use const by default — you won't reassign most things
const API_URL = "https://api.example.com";
const users = [];          // const is fine: we mutate the array, not the binding
const config = {};         // const is fine: we'll mutate properties

// ✅ Use let when you KNOW you'll reassign
let count = 0;
count++;                   // reassigning — let is correct here

let winner = "unknown";
if (score > 90) winner = "Alice"; // reassigning — let is correct

// ❌ Never use var
var legacy = "avoid me"; // scoping surprises, hoisting bugs, redeclaration
✅ const → let → never var
const doesn't mean "immutable value" — it means "immutable binding." The variable can't be reassigned to a new value, but the value itself can still be mutated if it's an object or array.

2 Template Literals

Backtick strings (`...`) let you embed any JS expression with ${ } — no more string concatenation.

❌ ES5 Concatenation
var name = "Alice";
var age = 30;
var msg = "Hello, " + name +
  "! You are " + age +
  " years old.";
console.log(msg);
Hello, Alice! You are 30 years old.
✅ ES6 Template Literal
const name = "Alice";
const age = 30;
const msg = `Hello, ${name}!
You are ${age} years old.`;
console.log(msg);
Hello, Alice! You are 30 years old.
Example 2 — Multi-line strings: just press Enter inside backticks
const poem = `Roses are red,
Violets are blue,
ES6 is awesome,
And so are you.`;
console.log(poem);
Roses are red,
               Violets are blue,
               ES6 is awesome,
               And so are you.
Example 3 — Math expression inline: auto-computed total with tax
const price = 49.99;
const qty = 3;
console.log(`Subtotal: $${price * qty}`);
console.log(`Total with 20% tax: $${(price * qty * 1.2).toFixed(2)}`);
Subtotal: $149.97
               Total with 20% tax: $179.96
Example 4 — Ternary operator inside ${}: dynamic conditional text
const isLoggedIn = true;
const username = "Alice";

console.log(`${isLoggedIn ? `Welcome back, ${username}!` : 'Please log in.'}`);

const stock = 0;
console.log(`Status: ${stock > 0 ? 'In Stock' : 'Sold Out'}`);
Welcome back, Alice!
               Status: Sold Out
Example 5 — Function call inside ${}: transform the value on the fly
const name = "alice wonder";
const score = 0.876543;

console.log(`Player: ${name.toUpperCase()}`);
console.log(`Score: ${score.toFixed(2)} (${(score * 100).toFixed(1)}%)`);
console.log(`Words: ${name.split(' ').length}`);
Player: ALICE WONDER
               Score: 0.88 (87.7%)
               Words: 2
Example 6 — Array method inside ${}: join items into a readable list
const skills = ["JavaScript", "React", "CSS"];
const numbers = [4, 7, 2, 9, 1];

console.log(`Skills: ${skills.join(' · ')}`);
console.log(`Max score: ${Math.max(...numbers)}`);
console.log(`Sorted: ${[...numbers].sort((a,b) => a - b).join(', ')}`);
Skills: JavaScript · React · CSS
               Max score: 9
               Sorted: 1, 2, 4, 7, 9
Example 7 — Object property access: clean user profile string
const user = {
  name: "Bob",
  age: 28,
  address: { city: "Berlin", country: "Germany" }
};

console.log(`${user.name} is ${user.age} years old.`);
console.log(`Lives in: ${user.address.city}, ${user.address.country}`);
console.log(`Born around: ${new Date().getFullYear() - user.age}`);
Bob is 28 years old.
               Lives in: Berlin, Germany
               Born around: 1997
Example 8 — Nested template: .map() to build HTML list items
const items = ["Home", "About", "Contact"];

const navLinks = `
  <ul>
    ${items.map(item => `<li><a href="#${item.toLowerCase()}">${item}</a></li>`).join('\\n    ')}
  </ul>
`;
console.log(navLinks);
<ul>
                 <li><a href="#home">Home</a></li>
                 <li><a href="#about">About</a></li>
                 <li><a href="#contact">Contact</a></li>
             </ul>
Example 9 — Building an HTML card string with template literals
const product = { name: "Laptop Pro", price: 1299, rating: 4.8, stock: 5 };

const card = `
  <div class="product-card">
    <h2>${product.name}</h2>
    <p class="price">$${product.price.toLocaleString()}</p>
    <p class="rating">⭐ ${product.rating} / 5</p>
    <p class="stock">${product.stock > 0 ? `Only ${product.stock} left!` : 'Out of stock'}</p>
  </div>
`;
console.log(card);
<div class="product-card"> <h2>Laptop Pro</h2> <p>$1,299</p> <p>⭐ 4.8 / 5</p> <p>Only 5 left!</p> </div>
Example 10 — Multiple expressions in one string: full status report
const server = { name: "prod-01", uptime: 99.98, requests: 150432, errors: 12 };
const now = new Date();

const report = `
=== Server Report [${now.toLocaleDateString()}] ===
Server  : ${server.name.toUpperCase()}
Uptime  : ${server.uptime}%  ${server.uptime > 99 ? '✅ Healthy' : '⚠️ Degraded'}
Requests: ${server.requests.toLocaleString()}
Errors  : ${server.errors} (${((server.errors / server.requests) * 100).toFixed(3)}%)
`;
console.log(report);
=== Server Report [today] === Server: PROD-01 Uptime: 99.98% ✅ Healthy Requests: 150,432 Errors: 12 (0.008%)

3 Destructuring

Pull values out of objects and arrays into variables in one line — cleaner than repeated obj.property access.

Example 1 — Object destructuring: basic
const user = { name: "Alice", age: 30, role: "admin" };

// Without destructuring (old way)
const name1 = user.name;
const age1  = user.age;

// With destructuring (ES6) — one line, pulls all three
const { name, age, role } = user;
console.log(name, age, role);
Alice 30 admin
Example 2 — Object destructuring: rename while destructuring
const user = { name: "Alice", age: 30 };

// Rename 'name' to 'userName' and 'age' to 'userAge'
const { name: userName, age: userAge } = user;
console.log(userName); // "Alice"
console.log(userAge);  // 30
// console.log(name) — ReferenceError! 'name' wasn't declared
Alice
               30
Example 3 — Object destructuring: default values for missing keys
const settings = { theme: "dark" }; // 'fontSize' is missing

const { theme, fontSize = 16, language = "en" } = settings;
console.log(theme);    // "dark"    — from the object
console.log(fontSize); // 16       — default, key wasn't in the object
console.log(language); // "en"     — default, key wasn't in the object
dark
               16
               en
Example 4 — Object destructuring: rename AND set a default at once
const apiResponse = {}; // completely empty

// Rename 'name' → 'displayName', default to 'Anonymous'
const { name: displayName = "Anonymous", score: userScore = 0 } = apiResponse;
console.log(displayName); // "Anonymous"
console.log(userScore);   // 0
Anonymous
               0
Example 5 — Nested object destructuring: dive into nested objects
const user = {
  name: "Bob",
  address: {
    city: "Tokyo",
    zip: "100-0001",
    coords: { lat: 35.6762, lng: 139.6503 }
  }
};

const { name, address: { city, zip, coords: { lat } } } = user;
console.log(name, city, zip, lat);
Bob Tokyo 100-0001 35.6762
Example 6 — Destructuring in function parameters: clean function signatures
// Instead of: function greet(user) { user.name, user.age... }
function greet({ name, age, role = "guest" }) {
  console.log(`Hi ${name}! Age: ${age}, Role: ${role}`);
}

greet({ name: "Alice", age: 30, role: "admin" });
greet({ name: "Charlie", age: 22 }); // role defaults to "guest"
Hi Alice! Age: 30, Role: admin
               Hi Charlie! Age: 22, Role: guest
Example 7 — API response destructuring: real-world pattern
const response = {
  status: 200,
  data: {
    users: ["Alice", "Bob", "Charlie"],
    total: 3,
    page: 1
  }
};

const { status, data: { users, total, page } } = response;
console.log(`Status: ${status}`);
console.log(`Page ${page}: ${users.join(', ')} (${total} total)`);
Status: 200
               Page 1: Alice, Bob, Charlie (3 total)
Example 8 — Array destructuring: basic positional unpacking
const colors = ["red", "green", "blue"];
const rgb = [255, 128, 0];

const [first, second, third] = colors;
const [r, g, b] = rgb;

console.log(first, second, third);
console.log(`rgb(${r}, ${g}, ${b})`);
red green blue
               rgb(255, 128, 0)
Example 9 — Array destructuring: skip elements with commas
const data = ["Alice", "28", "Engineer", "Berlin", "Germany"];

// Skip index 1 and 3 — only grab what you need
const [name, , role, , country] = data;
console.log(name, role, country);

// Skip first two items, grab only the third
const [,, profession] = data;
console.log(profession);
Alice Engineer Germany
               Engineer
Example 10 — Array destructuring: default values for missing positions
const partial = [10]; // only one item

const [a, b = 0, c = 100] = partial;
console.log(a); // 10  — from the array
console.log(b); // 0   — default, position 1 was undefined
console.log(c); // 100 — default, position 2 was undefined
10
               0
               100
Example 11 — Array destructuring: swap two variables without a temp variable
let x = 1;
let y = 99;
console.log(`Before: x=${x}, y=${y}`);

// Swap in one line — no temp variable needed!
[x, y] = [y, x];
console.log(`After:  x=${x}, y=${y}`);
Before: x=1, y=99
               After: x=99, y=1
Example 12 — Array destructuring with rest: head and tail of a list
const queue = ["Alice", "Bob", "Charlie", "Diana", "Eve"];

// First person out, rest still waiting
const [first, ...remaining] = queue;
console.log(`Serving: ${first}`);
console.log(`Waiting: ${remaining.join(', ')}`);

// Grab first two, rest is the tail
const [head, second, ...tail] = queue;
console.log(`Top 2: ${head}, ${second} | Tail: ${tail.join(', ')}`);
Serving: Alice
               Waiting: Bob, Charlie, Diana, Eve
               Top 2: Alice, Bob | Tail: Charlie, Diana, Eve
Example 13 — Mixed: array of objects destructured in one line
const users = [
  { name: "Alice", score: 98 },
  { name: "Bob",   score: 87 },
  { name: "Carol", score: 92 }
];

// Destructure first item's name, second item's name
const [{ name: gold }, { name: silver }, { name: bronze }] = users;
console.log(`🥇 ${gold}  🥈 ${silver}  🥉 ${bronze}`);

// Map + destructure in one go
users.forEach(({ name, score }) => console.log(`${name}: ${score}`));
🥇 Alice 🥈 Bob 🥉 Carol
               Alice: 98
               Bob: 87
               Carol: 92

4 Object Shorthand

When a variable name matches the key name, you can drop the repetition. Same goes for methods and computed keys.

❌ Verbose (ES5)
const name = "Alice";
const age  = 30;
const role = "admin";

const user = {
  name: name,
  age: age,
  role: role
};
console.log(user);
{ name: 'Alice', age: 30, role: 'admin' }
✅ Property Shorthand (ES6)
const name = "Alice";
const age  = 30;
const role = "admin";

const user = { name, age, role };
// key === variable name → just write it once
console.log(user);
{ name: 'Alice', age: 30, role: 'admin' }
❌ Method verbose (ES5)
const calculator = {
  add: function(a, b) {
    return a + b;
  },
  sub: function(a, b) {
    return a - b;
  }
};
console.log(calculator.add(5, 3));
8
✅ Method shorthand (ES6)
const calculator = {
  add(a, b) {
    return a + b;
  },
  sub(a, b) {
    return a - b;
  }
};
console.log(calculator.add(5, 3));
8
Example 3 — Computed property key: dynamic key from a variable
const fieldName = "email";
const value = "alice@example.com";

// [expression] as a key — computed at runtime
const formData = { [fieldName]: value };
console.log(formData); // { email: 'alice@example.com' }

const key = "score";
const metrics = { [key]: 95, [`${key}_max`]: 100 };
console.log(metrics); // { score: 95, score_max: 100 }
{ email: 'alice@example.com' }
               { score: 95, score_max: 100 }
Example 4 — Dynamic key with template literal: build keys programmatically
const id = 42;
const userData = { name: "Alice", score: 98 };

const cache = { [`user_${id}`]: userData };
console.log(cache); // { user_42: { name: 'Alice', score: 98 } }

// Building namespaced config keys
const env = "prod";
const config = {
  [`${env}_api_url`]: "https://api.example.com",
  [`${env}_timeout`]: 5000
};
console.log(config);
{ user_42: { name: 'Alice', score: 98 } }
               { prod_api_url: 'https://api.example.com', prod_timeout: 5000 }
Example 5 — Arrow function returning an object: WRONG (curly brace ambiguity!)
// ❌ JS interprets {} as a function BODY, not an object literal
const getUser = () => { name: "Alice", age: 30 };
console.log(getUser()); // undefined — treated as labeled statement!

// Even this looks right but isn't:
const names = ["Alice", "Bob"];
const wrong = names.map(name => { id: name });
console.log(wrong); // [undefined, undefined]
undefined
[undefined, undefined]
Example 6 — Arrow function returning an object: RIGHT (wrap in parens)
// ✅ Wrap the object in () so JS knows it's an expression
const getUser = () => ({ name: "Alice", age: 30 });
console.log(getUser()); // { name: 'Alice', age: 30 }

const names = ["Alice", "Bob"];
const users = names.map((name, i) => ({ id: i + 1, name }));
console.log(users);
{ name: 'Alice', age: 30 }
               [ { id: 1, name: 'Alice' }, { id: 2, name: 'Bob' } ]
Example 7 — Combining shorthand + destructuring: classic React-style pattern
// React-style: receive props, destructure, return shorthand object
function processUser({ name, age, email }) {
  const isAdult = age >= 18;
  const domain = email.split("@")[1];

  // Return new object using property shorthand
  return { name, age, isAdult, domain };
}

const result = processUser({ name: "Alice", age: 30, email: "alice@gmail.com" });
console.log(result);
{ name: 'Alice', age: 30, isAdult: true, domain: 'gmail.com' }
Example 8 — Merging objects with spread: defaults + overrides pattern
const defaults = { theme: "light", lang: "en", fontSize: 14, notifications: true };
const userPrefs = { theme: "dark", fontSize: 18 };

// Spread: later keys overwrite earlier ones
const finalConfig = { ...defaults, ...userPrefs };
console.log(finalConfig);

// Add extra property on top
const sessionConfig = { ...finalConfig, sessionId: "abc123" };
console.log(sessionConfig.sessionId, sessionConfig.theme);
{ theme: 'dark', lang: 'en', fontSize: 18, notifications: true }
               abc123 dark

function createProfileCard({ name, age, role, skills, isActive }) {
  const badgeClass = isActive ? "badge-active" : "badge-inactive";
  const badgeText  = isActive ? "● Active" : "○ Inactive";
  const skillTags  = skills.map(s => `<span class="skill-tag">${s}</span>`).join("");

  return `
    <div class="demo-card">
      <h3>${name}</h3>
      <div class="role">${role}</div>
      <span class="${badgeClass}">${badgeText}</span>
      <p style="font-size:13px;color:#7a8599;margin:8px 0 6px">Age: ${age}</p>
      <div class="skills">${skillTags}</div>
    </div>
  `;
}

const user = {
  name: "Alice Wonder",
  age: 30,
  role: "Senior Engineer",
  skills: ["JavaScript", "React", "Node.js", "CSS"],
  isActive: true
};

document.getElementById("profile-output").innerHTML = createProfileCard(user);

function buildConfig({
  theme         = 'dark',
  lang          = 'en',
  fontSize      = 16,
  notifications = true
} = {}) {
  const displayName = `${theme.charAt(0).toUpperCase() + theme.slice(1)} Theme`;
  const timestamp   = new Date().toISOString();
  const base        = { version: '1.0.0', env: 'browser' };

  const config = { ...base, theme, lang, fontSize, notifications, displayName, timestamp };

  console.log(`
╔══════════════════════════════════╗
  CONFIG BUILT: ${displayName}
  Lang      : ${lang.toUpperCase()}
  Font Size : ${fontSize}px
  Alerts    : ${notifications ? 'Enabled ✅' : 'Disabled ❌'}
  Built At  : ${timestamp}
  Version   : ${base.version}
╚══════════════════════════════════╝`);

  return config;
}

// Test 2: with all defaults (empty call)

console.log(myConfig);
