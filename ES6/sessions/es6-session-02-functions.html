ES6 Session 2: Arrow Functions, Defaults, Rest/Spread & Classes

Functions got a massive overhaul in ES6. This session is all about the new syntax — less boilerplate, more clarity, more power.

1 Arrow Functions

=> — A shorter syntax for writing functions. Less typing, implicit return, and lexical this.

    return n * 2;
};
✅ ES6 — Arrow Function
const double = (n) => n * 2;

this DIFFERENCE — regular vs arrow in setTimeout
const timer = {
    seconds: 0,

    // ❌ Regular function — `this` is the global/window object
    startBroken() {
        setTimeout(function() {
            this.seconds++;        // `this` is NOT timer!
            console.log(this.seconds); // NaN
        }, 1000);
    },

    // ✅ Arrow function — `this` inherits from startFixed()
    startFixed() {
        setTimeout(() => {
            this.seconds++;        // `this` IS timer ✓
            console.log(this.seconds); // 1
        }, 1000);
    }
};
startBroken() → NaN (this = window)
startFixed() → 1 (this = timer) ✓
Arrow functions don't have their own this. They capture this from where they're defined — this is called lexical this.

2 Default Parameters

param = value — Assign a fallback value right in the function signature.

❌ ES5 — || Fallback (buggy with falsy values)
function greet(name) {
    name = name || 'World';
    // Bug: greet('') → 'World'
    return 'Hello, ' + name;
}
✅ ES6 — Default Parameter
const greet = (name = 'World') => {
    // Only triggers on undefined
    return `Hello, ${name}`;
};
BASIC DEFAULT PARAMETER
const greet = (name = 'World') => `Hello, ${name}!`;

console.log(greet('Ada'));
console.log(greet());
console.log(greet(undefined));
"Hello, Ada!"
"Hello, World!"
"Hello, World!"
DEFAULT FROM ANOTHER PARAMETER
const box = (width, height = width) => ({ width, height });
const range = (start, end, step = 1) => ({ start, end, step });

console.log(box(100));          // square — height defaults to width
console.log(box(100, 50));
console.log(range(0, 10));
console.log(range(0, 10, 2));

{ width: 100, height: 50 }

3 Rest & Spread

... — Three dots. Same syntax, two jobs: rest collects, spread expands.

REST — ...args collects all arguments into an array
const sum = (...nums) => nums.reduce((a, b) => a + b, 0);

console.log(sum(1, 2));
console.log(sum(1, 2, 3, 4, 5));
console.log(sum(10, 20, 30));
3
15
60
REST — named param + ...rest for the remainder
const log = (level, ...messages) => {
    return `[${level.toUpperCase()}] ${messages.join(' ')}`;
};

console.log(log('info', 'Server', 'started', 'on', 'port', 3000));
console.log(log('error', 'File not found'));
console.log(log('warn', 'Disk', 'space', 'low'));
"[INFO] Server started on port 3000"
"[ERROR] File not found"
"[WARN] Disk space low"
REST — array destructuring [first, ...rest]
const [first, second, ...remaining] = [10, 20, 30, 40, 50];

console.log(first);
console.log(second);
console.log(remaining);
10
20
[30, 40, 50]
REST — object destructuring { name, ...others }
const user = { name: 'Ada', age: 25, role: 'dev', city: 'London' };
const { name, ...profile } = user;

console.log(name);
console.log(profile);
"Ada"
{ age: 25, role: "dev", city: "London" }
SPREAD — expand array into function arguments
const nums = [3, 7, 1, 9, 4];

console.log(Math.max(...nums));
console.log(Math.min(...nums));

// Without spread: Math.max(3, 7, 1, 9, 4)
// Spread does that automatically!

1
SPREAD — copy an array (shallow clone)
const original = [1, 2, 3];
const copy = [...original];

copy.push(4);

console.log(original);  // untouched!
console.log(copy);      // has the new item
[1, 2, 3]
[1, 2, 3, 4]
SPREAD — merge arrays
const front = ['🥇', '🥈'];
const back  = ['🥉', '🏅'];

const all   = [...front, ...back];
const mixed = [...back, '⭐', ...front];

console.log(mixed);

["🥉", "🏅", "⭐", "🥇", "🥈"]
SPREAD — add items without mutating
const todos = ['eat', 'code'];

const added   = [...todos, 'sleep'];       // append
const prepend = ['wake up', ...todos];     // prepend

SPREAD — copy an object (shallow clone)

const clone  = { ...config };

clone.lang = 'fr';

console.log(config);  // untouched
console.log(clone);

SPREAD — override a single property immutably
const user = { name: 'Ada', age: 25, role: 'dev' };

console.log(user);

⚠️ SHALLOW COPY GOTCHA — nested objects are shared!
const original = {
    name: 'Ada',
    scores: [90, 85, 92]    // nested array (reference!)
};

const clone = { ...original };
clone.name = 'Tim';              // ✓ does not affect original
clone.scores.push(100);          // ✗ DOES affect original!

console.log(original.name);
console.log(original.scores);     // oops, 100 is here too!
"Ada"
[90, 85, 92, 100] ← mutation leaked!
Spread only copies one level deep. For nested objects/arrays, use structuredClone(obj) or JSON.parse(JSON.stringify(obj)).

4 Classes

class — Syntactic sugar over prototypes. Cleaner, familiar OOP syntax.

❌ ES5 — Constructor Function
function Person(name, age) {
    this.name = name;
    this.age = age;
}
Person.prototype.greet = function() {
    return 'Hi, I am ' + this.name;
};
✅ ES6 — Class
class Person {
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
    greet() {
        return `Hi, I am ${this.name}`;
    }
}
BASIC CLASS + CONSTRUCTOR

SUPER.METHOD() — extend parent instead of replacing
class Component {
    render() {
        return '<div>';
    }
}

class Button extends Component {
    constructor(label) {
        super();
        this.label = label;
    }
    render() {
        const wrapper = super.render();   // call parent's render
        return `${wrapper}<button>${this.label}</button></div>`;
    }
}

console.log(new Button('Click Me').render());
"<div><button>Click Me</button></div>"
PRIVATE # FIELDS — truly inaccessible from outside
class BankAccount {
    #balance = 0;  // private field

    constructor(owner, initial = 0) {
        this.owner = owner;
        this.#balance = initial;
    }

    deposit(amount) {
        if (amount > 0) this.#balance += amount;
        return this;
    }

    withdraw(amount) {
        if (amount > 0 && amount <= this.#balance) {
            this.#balance -= amount;
        }
        return this;
    }

    get balance() { return this.#balance; }
}

const acct = new BankAccount('Ada', 1000);
acct.deposit(500).withdraw(200);
console.log(acct.balance);
// acct.#balance → SyntaxError! Can't access from outside

REAL WORLD — ShoppingCart class
class ShoppingCart {
    #items = [];

    addItem(name, price, qty = 1) {
        this.#items.push({ name, price, qty });
        return this;
    }

    get total() {
        return this.#items.reduce((sum, i) => sum + i.price * i.qty, 0);
    }

    get count() {
        return this.#items.reduce((sum, i) => sum + i.qty, 0);
    }

    summary() {
        return this.#items
            .map(i => `${i.name} x${i.qty} = $${(i.price * i.qty).toFixed(2)}`)
            .join('\n');
    }
}

const cart = new ShoppingCart();

console.log(`Items: ${cart.count} | Total: $${cart.total.toFixed(2)}`);

🏆 Challenge 1: Shopping Cart Class

Build a ShoppingCart class with:

  • addItem(name, price, qty) — adds item, returns this for chaining
  • removeItem(name) — removes item by name
  • getTotal() — uses .reduce() to sum price × qty
  • getSummary() — returns a template literal summary string

Expected output from demo code below:

🛒 Shopping Cart ──────────────────── Laptop x1 = $999.99 Mouse x2 = $49.98 Keyboard x1 = $75.00 ──────────────────── Total: $1,124.97 (4 items)
Hint

Use #items = [] for a private array. In removeItem, use .filter() to keep everything except the named item. In getTotal, use .reduce((sum, item) => sum + item.price * item.qty, 0). For the summary, use .map() + .join('\n').

Solution
class ShoppingCart {
    #items = [];

    addItem(name, price, qty = 1) {
        this.#items.push({ name, price, qty });
        return this;
    }

    removeItem(name) {
        this.#items = this.#items.filter(i => i.name !== name);
        return this;
    }

    getTotal() {
        return this.#items.reduce((sum, i) => sum + i.price * i.qty, 0);
    }

    getSummary() {
        const itemCount = this.#items.reduce((s, i) => s + i.qty, 0);
        const lines = this.#items
            .map(i => `${i.name} x${i.qty} = $${(i.price * i.qty).toFixed(2)}`)
            .join('\n');
        const total = this.getTotal().toLocaleString('en-US', {
            minimumFractionDigits: 2
        });
        return `🛒 Shopping Cart\n${'─'.repeat(20)}\n${lines}\n${'─'.repeat(20)}\nTotal: $${total} (${itemCount} items)`;
    }
}

// Demo
const cart = new ShoppingCart();
cart.addItem('Laptop', 999.99)
    .addItem('Mouse', 24.99, 2)
    .addItem('Keyboard', 75.00);

console.log(cart.getSummary());

🏆 Challenge 2: Function Toolkit

Write 5 arrow utility functions using rest/spread:

  • double(...nums) — returns array with each number doubled
  • evens(...nums) — returns only even numbers
  • sum(...nums) — returns total of all numbers
  • merge(...objects) — merges all objects using spread
  • pick(obj, ...keys) — returns new object with only the listed keys

Expected output:

double(1,2,3,4,5) → [2, 4, 6, 8, 10] evens(1,2,3,4,5,6) → [2, 4, 6] sum(10,20,30) → 60 merge({a:1},{b:2}) → {a:1, b:2} pick(user,'name','age') → {name:"Ada", age:25}
Hint

double: use nums.map(n => n * 2). evens: use nums.filter(n => n % 2 === 0). sum: use nums.reduce(). merge: use objects.reduce((acc, obj) => ({...acc, ...obj}), {}). pick: use Object.fromEntries(keys.filter(k => k in obj).map(k => [k, obj[k]])).

Solution
// 1. Double all numbers
const double = (...nums) => nums.map(n => n * 2);

// 2. Filter to evens only
const evens = (...nums) => nums.filter(n => n % 2 === 0);

// 3. Sum all numbers
const sum = (...nums) => nums.reduce((a, b) => a + b, 0);

// 4. Merge all objects
const merge = (...objects) => objects.reduce((acc, obj) => ({ ...acc, ...obj }), {});

// 5. Pick specific keys from object
const pick = (obj, ...keys) => {
    return Object.fromEntries(
        keys.filter(k => k in obj).map(k => [k, obj[k]])
    );
};

// Test
console.log(double(1, 2, 3, 4, 5));
console.log(evens(1, 2, 3, 4, 5, 6));
console.log(sum(10, 20, 30));
console.log(merge({ a: 1 }, { b: 2 }, { c: 3 }));

const user = { name: 'Ada', age: 25, role: 'dev', city: 'London' };
console.log(pick(user, 'name', 'age'));
