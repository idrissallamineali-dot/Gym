🔢 Number Katas

20 challenges focused on numbers and math logic. Each kata has a constraint that forces you to think differently.

01Easy

Is Even Without Modulo

Write isEven(n) that checks if a number is even.

isEven(4)   // true
isEven(7)   // false
Do NOT use the % operator. Use bitwise AND (&).

Solution

const isEven = n => (n & 1) === 0;
// Bitwise AND with 1: even numbers have 0 as last bit, odd have 1.
02Easy

Factorial (Recursive)

Write factorial(n).

factorial(5)   // 120
factorial(0)   // 1
Use recursion. No loops.

Solution

function factorial(n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}
03Easy

Factorial (Iterative)

Write factorialLoop(n).

factorialLoop(5)   // 120
Use a for loop. No recursion.

Solution

function factorialLoop(n) {
    let result = 1;
    for (let i = 2; i <= n; i++) {
        result *= i;
    }
    return result;
}
04Easy

Fibonacci N-th

Write fib(n) that returns the n-th Fibonacci number (0-indexed).

fib(0)   // 0
fib(1)   // 1
fib(6)   // 8  (0,1,1,2,3,5,8)
Use a for loop with only two variables (prev, curr). No arrays, no recursion.

Solution

function fib(n) {
    if (n <= 0) return 0;
    let prev = 0, curr = 1;
    for (let i = 2; i <= n; i++) {
        [prev, curr] = [curr, prev + curr];
    }
    return curr;
}

const digitSum = n =>

06Medium

Is Prime

Write isPrime(n).

isPrime(2)    // true
isPrime(17)   // true
isPrime(4)    // false
isPrime(1)    // false
Only loop up to Math.sqrt(n). Use a for loop. Use guard clauses for edge cases.

Solution

function isPrime(n) {
    if (n < 2) return false;
    if (n === 2) return true;
    if (n % 2 === 0) return false;
    for (let i = 3; i <= Math.sqrt(n); i += 2) {
        if (n % i === 0) return false;
    }
    return true;
}
07Medium

Prime Factors

Write primeFactors(n).

primeFactors(12)   // [2, 2, 3]
primeFactors(100)  // [2, 2, 5, 5]
Use a while loop and start dividing from 2 upwards.

Solution

function primeFactors(n) {
    const factors = [];
    let d = 2;
    while (n > 1) {
        while (n % d === 0) {
            factors.push(d);
            n /= d;
        }
        d++;
    }
    return factors;
}

Write gcd(a, b).

Solution

function gcd(a, b) {

Write lcm(a, b).

lcm(3, 7)    // 21

Solution

function gcd(a, b) { return b === 0 ? a : gcd(b, a % b); }
const lcm = (a, b) => (a * b) / gcd(a, b);
10Medium

Power Without Math.pow

Write power(base, exp). Handle exp = 0.

power(2, 10)  // 1024
power(5, 0)   // 1
power(3, 3)   // 27
Use recursion. No loops, no Math.pow, no ** operator.

Solution

function power(base, exp) {
    if (exp === 0) return 1;
    return base * power(base, exp - 1);
}
11Hard

Integer to Roman

Write toRoman(num). Handle 1–3999.

toRoman(3)     // "III"
toRoman(49)    // "XLIX"
toRoman(1994)  // "MCMXCIV"
Use an array of [value, symbol] pairs and a while loop.

Solution

function toRoman(num) {
    const pairs = [
        [1000,"M"],[900,"CM"],[500,"D"],[400,"CD"],
        [100,"C"],[90,"XC"],[50,"L"],[40,"XL"],
        [10,"X"],[9,"IX"],[5,"V"],[4,"IV"],[1,"I"]
    ];
    let result = "";
    for (const [value, symbol] of pairs) {
        while (num >= value) {
            result += symbol;
            num -= value;
        }
    }
    return result;
}
12Hard

Roman to Integer

Write fromRoman(str).

fromRoman("III")      // 3
fromRoman("XLIX")     // 49
fromRoman("MCMXCIV")  // 1994
Use a lookup object. If current symbol is less than the next, subtract it instead of adding.

Solution

function fromRoman(str) {
    const map = { I:1, V:5, X:10, L:50, C:100, D:500, M:1000 };
    let total = 0;
    for (let i = 0; i < str.length; i++) {
        if (map[str[i]] < map[str[i + 1]]) {
            total -= map[str[i]];
        } else {
            total += map[str[i]];
        }
    }
    return total;
}
13Medium

Decimal to Binary

Write toBinary(n).

toBinary(10)   // "1010"
toBinary(255)  // "11111111"
Do NOT use .toString(2). Use a while loop dividing by 2 and collecting remainders.

Solution

function toBinary(n) {
    if (n === 0) return "0";
    let result = "";
    while (n > 0) {
        result = (n % 2) + result;
        n = Math.floor(n / 2);
    }
    return result;
}
14Medium

Binary to Decimal

Write fromBinary(str).

fromBinary("1010")      // 10
fromBinary("11111111")  // 255

Solution

const fromBinary = str =>
    str.split("").reduce((total, bit) => total * 2 + Number(bit), 0);
15Hard

Number to Words (0–999)

Write toWords(n).

toWords(0)     // "zero"
toWords(42)    // "forty two"
toWords(115)   // "one hundred fifteen"
toWords(999)   // "nine hundred ninety nine"
Use lookup arrays for ones, teens, and tens. No external libraries.

Solution

function toWords(n) {
    if (n === 0) return "zero";
    const ones = ["","one","two","three","four","five","six","seven","eight","nine",
                  "ten","eleven","twelve","thirteen","fourteen","fifteen","sixteen",
                  "seventeen","eighteen","nineteen"];
    const tens = ["","","twenty","thirty","forty","fifty","sixty","seventy","eighty","ninety"];

    let result = "";
    if (n >= 100) {
        result += ones[Math.floor(n / 100)] + " hundred ";
        n %= 100;
    }
    if (n >= 20) {
        result += tens[Math.floor(n / 10)] + " ";
        n %= 10;
    }
    if (n > 0) {
        result += ones[n];
    }
    return result.trim();
}
16Easy

Collatz Conjecture

Write collatz(n) that counts steps to reach 1. If even: n/2. If odd: 3n+1.

collatz(1)   // 0
collatz(6)   // 8
collatz(12)  // 9
Use a while loop and a counter.

Solution

function collatz(n) {
    let steps = 0;
    while (n !== 1) {
        n = n % 2 === 0 ? n / 2 : 3 * n + 1;
        steps++;
    }
    return steps;
}
17Medium

Sum of Multiples

Write sumMultiples(n). Sum all multiples of 3 or 5 below n.

sumMultiples(10)   // 23
sumMultiples(20)   // 78
Use Array.from() to generate the range, then filter + reduce. No for loops.

Solution

const sumMultiples = n =>
    Array.from({ length: n }, (_, i) => i)
         .filter(i => i % 3 === 0 || i % 5 === 0)
         .reduce((sum, i) => sum + i, 0);
18Medium

Perfect Number

Write isPerfect(n). A perfect number equals the sum of its proper divisors (exclude itself).

isPerfect(6)     // true  (1+2+3 = 6)
isPerfect(28)    // true  (1+2+4+7+14 = 28)
isPerfect(12)    // false
Use a for loop up to n/2 to find divisors.

Solution

function isPerfect(n) {
    if (n < 2) return false;
    let sum = 1;
    for (let i = 2; i <= n / 2; i++) {
        if (n % i === 0) sum += i;
    }
    return sum === n;
}
19Medium

Armstrong Number

Write isArmstrong(n). An Armstrong number equals the sum of its digits raised to the power of the digit count.

isArmstrong(153)   // true  (1³+5³+3³ = 153)
isArmstrong(370)   // true
isArmstrong(10)    // false
Use String(), split(), map(), reduce(). No manual loops.

Solution

function isArmstrong(n) {
    const digits = String(n).split("");
    const len = digits.length;
    const sum = digits.map(Number).reduce((acc, d) => acc + d ** len, 0);
    return sum === n;
}
20Medium

Sum of Even Fibonacci

Write evenFibSum(limit). Sum all even Fibonacci numbers below limit.

evenFibSum(100)   // 44  (2+8+34)
evenFibSum(60)    // 44
evenFibSum(10)    // 10  (2+8)
Use a while loop and an accumulator. Generate Fibonacci inline — no arrays.

Solution

function evenFibSum(limit) {
    let sum = 0;
    let prev = 0, curr = 1;
    while (curr < limit) {
        if (curr % 2 === 0) sum += curr;
        [prev, curr] = [curr, prev + curr];
    }
    return sum;
}
