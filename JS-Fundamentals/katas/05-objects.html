🧱 Object Katas

20 challenges focused on object manipulation, destructuring, prototypes, and patterns. Constraints force you to use specific techniques.

01Easy

Pick

Write pick(obj, keys). Return a new object with only the specified keys.

pick({ a:1, b:2, c:3 }, ["a","c"])   // { a: 1, c: 3 }
Use reduce.

Solution

const pick = (obj, keys) =>
    keys.reduce((result, key) => {
        if (key in obj) result[key] = obj[key];
        return result;
    }, {});
02Easy

Omit

Write omit(obj, keys). Return a new object WITHOUT the specified keys.

omit({ a:1, b:2, c:3 }, ["b"])   // { a: 1, c: 3 }
Use Object.entries(), filter, and Object.fromEntries().

Solution

const omit = (obj, keys) =>
    Object.fromEntries(
        Object.entries(obj).filter(([key]) => !keys.includes(key))
    );
03Easy

Invert

Write invert(obj). Swap keys and values.

invert({ a: 1, b: 2, c: 3 })   // { "1": "a", "2": "b", "3": "c" }
Use Object.fromEntries and Object.entries with map.

Solution

const invert = obj =>
    Object.fromEntries(Object.entries(obj).map(([k, v]) => [v, k]));
04Hard

Deep Merge

Write deepMerge(target, source). Recursively merge — nested objects are merged, not overwritten.

deepMerge(
    { a: 1, b: { x: 1, y: 2 } },
    { b: { y: 99, z: 3 }, c: 4 }
)
// { a: 1, b: { x: 1, y: 99, z: 3 }, c: 4 }
Use recursion and typeof checks. Do NOT mutate the originals.

Solution

function deepMerge(target, source) {
    const result = { ...target };
    for (const key of Object.keys(source)) {
        if (
            source[key] && typeof source[key] === "object" && !Array.isArray(source[key])
            && target[key] && typeof target[key] === "object" && !Array.isArray(target[key])
        ) {
            result[key] = deepMerge(target[key], source[key]);
        } else {
            result[key] = source[key];
        }
    }
    return result;
}
05Hard

Flatten Object

Write flattenObj(obj). Flatten nested objects into dot-notation keys.

flattenObj({ a: { b: { c: 1 } }, d: 2 })
// { "a.b.c": 1, "d": 2 }
Use recursion. Build the prefix as you recurse deeper.

Solution

function flattenObj(obj, prefix = "", result = {}) {
    for (const [key, value] of Object.entries(obj)) {
        const newKey = prefix ? `${prefix}.${key}` : key;
        if (value && typeof value === "object" && !Array.isArray(value)) {
            flattenObj(value, newKey, result);
        } else {
            result[newKey] = value;
        }
    }
    return result;
}
06Hard

Unflatten Object

Write unflattenObj(obj). Reverse of flatten.

unflattenObj({ "a.b.c": 1, "d": 2 })
// { a: { b: { c: 1 } }, d: 2 }
Use reduce and split("."). Build nested objects with a loop.

Solution

function unflattenObj(obj) {
    return Object.entries(obj).reduce((result, [path, value]) => {
        const keys = path.split(".");
        let current = result;
        for (let i = 0; i < keys.length - 1; i++) {
            if (!current[keys[i]]) current[keys[i]] = {};
            current = current[keys[i]];
        }
        current[keys[keys.length - 1]] = value;
        return result;
    }, {});
}
07Hard

Deep Clone

Write deepClone(obj).

const original = { a: 1, b: { c: [2, 3] } };
const clone = deepClone(original);
clone.b.c.push(4);
original.b.c  // [2, 3] — untouched!
Do NOT use structuredClone or JSON.parse/stringify. Use recursion.

Solution

function deepClone(obj) {
    if (obj === null || typeof obj !== "object") return obj;
    if (Array.isArray(obj)) return obj.map(item => deepClone(item));
    const clone = {};
    for (const [key, value] of Object.entries(obj)) {
        clone[key] = deepClone(value);
    }
    return clone;
}
08Hard

Deep Equal

Write deepEqual(a, b).

deepEqual({ a: { b: 1 } }, { a: { b: 1 } })  // true
deepEqual({ a: { b: 1 } }, { a: { b: 2 } })  // false
deepEqual([1, [2]], [1, [2]])                  // true
Use recursion and Object.keys. Handle arrays, objects, and primitives.

Solution

function deepEqual(a, b) {
    if (a === b) return true;
    if (a === null || b === null) return false;
    if (typeof a !== "object" || typeof b !== "object") return false;
    if (Array.isArray(a) !== Array.isArray(b)) return false;
    const keysA = Object.keys(a), keysB = Object.keys(b);
    if (keysA.length !== keysB.length) return false;
    return keysA.every(key => deepEqual(a[key], b[key]));
}

filterObj({ a: 1, b: 2, c: 3, d: 4 }, v => v > 2)

11Easy

To Query String

Write toQS(obj).

toQS({ name: "Idris", age: 25, city: "Paris" })
// "name=Idris&age=25&city=Paris"
Use entries, map, join. One chain.

Solution

const toQS = obj =>
    Object.entries(obj).map(([k, v]) => `${k}=${v}`).join("&");
12Medium

From Query String

Write fromQS(str).

fromQS("name=Idris&age=25&city=Paris")
// { name: "Idris", age: "25", city: "Paris" }
Use split, map, and Object.fromEntries.

Solution

const fromQS = str =>
    Object.fromEntries(str.split("&").map(pair => pair.split("=")));
13Medium

Group By

Write groupBy(arr, key).

groupBy([{c:"Paris",n:"A"},{c:"London",n:"B"},{c:"Paris",n:"C"}], "c")
// { Paris: [{...},{...}], London: [{...}] }
Use reduce only.

Solution

function groupBy(arr, key) {
    return arr.reduce((groups, item) => {
        const group = item[key];
        if (!groups[group]) groups[group] = [];
        groups[group].push(item);
        return groups;
    }, {});
}
14Medium

Count By

Write countBy(arr, fn). Count occurrences by a key function.

countBy([1,2,3,4,5], n => n % 2 === 0 ? "even" : "odd")
// { odd: 3, even: 2 }
Use reduce.

Solution

function countBy(arr, fn) {
    return arr.reduce((acc, item) => {
        const key = fn(item);
        acc[key] = (acc[key] || 0) + 1;
        return acc;
    }, {});
}

16Medium

Diff Objects

Write diff(a, b). Return object with only the properties that have different values.

diff({ x: 1, y: 2, z: 3 }, { x: 1, y: 99, z: 3 })
// { y: 99 }
Use Object.keys, filter, and reduce.

Solution

function diff(a, b) {
    return Object.keys(b)
        .filter(key => a[key] !== b[key])
        .reduce((result, key) => { result[key] = b[key]; return result; }, {});
}
17Medium

Path Get

Write get(obj, path). Access deeply nested properties safely via string path.

const data = { a: { b: { c: 42 } } };
get(data, "a.b.c")       // 42
get(data, "a.x.y")       // undefined (no crash)
Use split(".") and reduce with optional chaining.

Solution

const get = (obj, path) =>
    path.split(".").reduce((acc, key) => acc?.[key], obj);
18Medium

Path Set

Write set(obj, path, value). Set a deeply nested property, creating intermediate objects.

const obj = {};
set(obj, "a.b.c", 42);
obj  // { a: { b: { c: 42 } } }
Use split("."), a for loop, and bracket notation. Mutate the original.

Solution

function set(obj, path, value) {
    const keys = path.split(".");
    let current = obj;
    for (let i = 0; i < keys.length - 1; i++) {
        if (!current[keys[i]] || typeof current[keys[i]] !== "object") {
            current[keys[i]] = {};
        }
        current = current[keys[i]];
    }
    current[keys[keys.length - 1]] = value;
}
19Easy

Create Enum

Write createEnum(...values).

createEnum("RED", "GREEN", "BLUE")
// { RED: 0, GREEN: 1, BLUE: 2 }
Use reduce with the index parameter.

Solution

const createEnum = (...values) =>
    values.reduce((obj, val, i) => { obj[val] = i; return obj; }, {});
20Hard

Observable Object (Proxy)

Write observable(obj, onChange). When any property is set, call onChange(key, oldVal, newVal).

const logs = [];
const user = observable({ name: "Idris", age: 25 },
    (key, oldVal, newVal) => logs.push(`${key}: ${oldVal} → ${newVal}`)
);
user.name = "Updated";
user.age = 26;
logs  // ["name: Idris → Updated", "age: 25 → 26"]
Use Proxy with a set trap.

Solution

function observable(obj, onChange) {
    return new Proxy(obj, {
        set(target, key, value) {
            const oldVal = target[key];
            target[key] = value;
            onChange(key, oldVal, value);
            return true;
        }
    });
}
